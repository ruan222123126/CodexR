Project Context Handover: VS Code Extension wrapping "Codex CLI"

I am handing over the development of a VS Code Extension to you.

The extension creates a sidebar chat interface that acts as a GUI wrapper for *you* (the Codex CLI).

### Current Status

The extension is functional. It successfully:

1. Spawns the `codex` CLI process with arguments: `exec --dangerously-bypass-approvals-and-sandbox --skip-git-repo-check`.
2. Captures stdout/stderr.
3. Parses the raw output to separate "Thinking Process" from the "Final Answer".
4. Renders the result in a Webview using HTML/JS + `marked.js`.

### Key Challenges Solved (Do NOT regress on these):

1. **The "Chatty" Output Issue**: The raw CLI output mixes metadata, execution logs, and the final answer.
2. **The "Duplicate" Issue**: The CLI tends to repeat the final answer after printing "tokens used".
3. **The "Formatting" Issue**: Markdown line breaks require `breaks: true` in `marked.js`.
4. **The "Thinking" Visibility**: We want the reasoning/execution logs hidden in a collapsible `<details>` tag, not cluttering the view.

### Current Implementation: `src/CodexProvider.ts`

Review the logic below carefully, especially the `parseCodexOutput` method which handles the string splitting and cleaning.

```typescript
import * as vscode from 'vscode';
import * as cp from 'child_process';
import * as os from 'os';

export class CodexProvider implements vscode.WebviewViewProvider {
    public static readonly viewType = 'codex.chatView';
    private _view?: vscode.WebviewView;

    constructor(private readonly _extensionUri: vscode.Uri) { }

    public resolveWebviewView(
        webviewView: vscode.WebviewView,
        context: vscode.WebviewViewResolveContext,
        _token: vscode.CancellationToken,
    ) {
        this._view = webviewView;
        webviewView.webview.options = { enableScripts: true, localResourceRoots: [this._extensionUri] };
        webviewView.webview.html = this._getHtmlForWebview(webviewView.webview);

        webviewView.webview.onDidReceiveMessage(data => {
            if (data.type === 'userInput') {
                this.executeCodex(data.value);
            }
        });
    }

    private executeCodex(prompt: string) {
        this.postMessageToWebview('user', prompt);
        this.postMessageToWebview('system', 'Thinking...');

        let workspaceDir = os.homedir();
        if (vscode.workspace.workspaceFolders && vscode.workspace.workspaceFolders.length > 0) {
            workspaceDir = vscode.workspace.workspaceFolders[0].uri.fsPath;
        }

        const command = 'codex';
        const args = [
            'exec', 
            '--dangerously-bypass-approvals-and-sandbox', 
            '--skip-git-repo-check'
        ];

        const child = cp.spawn(command, args, {
            shell: true,
            cwd: workspaceDir,
            env: { ...process.env, LANG: "en_US.UTF-8", LC_ALL: "en_US.UTF-8" }
        });

        if (child.stdin) {
            child.stdin.write(prompt + "\n");
            child.stdin.end(); 
        }

        let outputBuffer = '';

        child.stdout?.on('data', (data: any) => {
            outputBuffer += data.toString();
        });

        child.stderr?.on('data', (data: any) => {
            outputBuffer += data.toString();
        });

        child.on('close', (code: number) => {
            // PARSING LOGIC: Separates thought from content
            const result = this.parseCodexOutput(outputBuffer);
            
            if (result.content || result.thought) {
                this._view?.webview.postMessage({ 
                    type: 'bot-complex', 
                    value: result 
                });
            } else if (code !== 0) {
                this.postMessageToWebview('error', `Process Failed. Raw:\n${outputBuffer}`);
            } else {
                this.postMessageToWebview('error', 'No response.');
            }
            
            this.postMessageToWebview('done', '');
        });
    }

    /**
     * Parsing Logic:
     * 1. Cuts off everything after "tokens used" to prevent duplication.
     * 2. Splits text at the last "codex" keyword.
     * - Before "codex": Thinking/Logs.
     * - After "codex": Final Response.
     */
    private parseCodexOutput(raw: string): { thought: string, content: string } {
        let text = raw.replace(/\x1b\[[0-9;]*m/g, ''); // Remove ANSI colors
        text = text.split(/tokens used/i)[0]; // Cut duplication

        // Clean headers
        text = text.replace(/^Reading prompt from stdin\.\.\./gm, '');
        text = text.replace(/mcp startup:.*$/gm, '');

        const splitKeyword = 'codex';
        const lastIndex = text.toLowerCase().lastIndexOf(splitKeyword);

        let thought = '';
        let content = '';

        if (lastIndex !== -1) {
            thought = text.substring(0, lastIndex).trim();
            content = text.substring(lastIndex + splitKeyword.length).trim();
        } else {
            content = text.trim();
        }

        // Clean up "thought" section
        const thinkingStart = thought.toLowerCase().indexOf('thinking');
        if (thinkingStart !== -1) {
            thought = thought.substring(thinkingStart).trim();
        } else {
            const userIndex = thought.toLowerCase().indexOf('user ');
            if (userIndex !== -1) {
                thought = thought.substring(userIndex).trim();
            }
        }
        
        thought = thought.replace(/^user\s+.*$/im, ''); 

        return { thought, content };
    }

    private postMessageToWebview(type: string, value: string) {
        this._view?.webview.postMessage({ type, value });
    }

    private _getHtmlForWebview(webview: vscode.Webview) {
        // HTML includes <details> for thinking process and marked.js with breaks:true
        return `<!DOCTYPE html>
            <html lang="en">
            <head>
                <meta charset="UTF-8">
                <script src="[https://cdn.jsdelivr.net/npm/marked/marked.min.js](https://cdn.jsdelivr.net/npm/marked/marked.min.js)"></script>
                <style>
                    /* Styles omitted for brevity, but they support dark mode and chat bubbles */
                    /* Important: details.thinking-block is used for the thinking process */
                </style>
            </head>
            <body>
                <div id="chat-container"></div>
                <div id="input-area">
                    <textarea id="input-box" placeholder="Ask Codex..."></textarea>
                    <button id="send-btn">Send</button>
                </div>
                <script>
                    const vscode = acquireVsCodeApi();
                    marked.setOptions({ breaks: true, gfm: true });
                    // ... Client side script that renders { thought, content }
                </script>
            </body>
            </html>`;
    }
}
```
